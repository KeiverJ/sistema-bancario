package com.example.repository;

import com.example.model.User;
import com.example.config.DatabaseConfig;

import java.util.*;
import java.util.concurrent.atomic.AtomicLong;

/**
 * UserRepository class simulating database operations.
 * Uses in-memory storage for demonstration purposes.
 * 
 * This class will be used to demonstrate different types of test doubles:
 * - Mocks: Verify behavior and interactions
 * - Stubs: Provide predefined responses
 * - Spies: Partial mocking of real objects
 * - Fakes: Simple working implementations
 */
public class UserRepository {

  private final Map<Long, User> users;
  private final AtomicLong idGenerator;
  private final DatabaseConfig databaseConfig;

  public UserRepository() {
    this.users = new HashMap<>();
    this.idGenerator = new AtomicLong(1);
    this.databaseConfig = DatabaseConfig.getInstance();
  }

  /**
   * Constructor that accepts a DatabaseConfig for dependency injection.
   * Useful for testing with different configurations.
   */
  public UserRepository(DatabaseConfig databaseConfig) {
    this.users = new HashMap<>();
    this.idGenerator = new AtomicLong(1);
    this.databaseConfig = databaseConfig;
  }

  /**
   * Saves a user to the repository.
   * 
   * @param user the user to save
   * @return the saved user with generated ID
   * @throws IllegalArgumentException if user is null or invalid
   */
  public User save(User user) {
    if (user == null) {
      throw new IllegalArgumentException("User cannot be null");
    }

    if (user.getUsername() == null || user.getUsername().trim().isEmpty()) {
      throw new IllegalArgumentException("Username cannot be null or empty");
    }

    if (user.getEmail() == null || !user.isValidEmail()) {
      throw new IllegalArgumentException("Valid email is required");
    }

    // Check if username already exists
    if (existsByUsername(user.getUsername())) {
      throw new IllegalArgumentException("Username already exists: " + user.getUsername());
    }

    // Generate ID if not set
    if (user.getId() == null) {
      user.setId(idGenerator.getAndIncrement());
    }

    users.put(user.getId(), user);
    return user;
  }

  /**
   * Finds a user by ID.
   * 
   * @param id the user ID
   * @return Optional containing the user if found, empty otherwise
   */
  public Optional<User> findById(Long id) {
    if (id == null) {
      return Optional.empty();
    }
    return Optional.ofNullable(users.get(id));
  }

  /**
   * Finds a user by username.
   * 
   * @param username the username to search for
   * @return Optional containing the user if found, empty otherwise
   */
  public Optional<User> findByUsername(String username) {
    if (username == null) {
      return Optional.empty();
    }

    return users.values().stream()
        .filter(user -> username.equals(user.getUsername()))
        .findFirst();
  }

  /**
   * Finds a user by email.
   * 
   * @param email the email to search for
   * @return Optional containing the user if found, empty otherwise
   */
  public Optional<User> findByEmail(String email) {
    if (email == null) {
      return Optional.empty();
    }

    return users.values().stream()
        .filter(user -> email.equals(user.getEmail()))
        .findFirst();
  }

  /**
   * Returns all users in the repository.
   * 
   * @return list of all users
   */
  public List<User> findAll() {
    return new ArrayList<>(users.values());
  }

  /**
   * Returns all active users.
   * 
   * @return list of active users
   */
  public List<User> findAllActive() {
    return users.values().stream()
        .filter(User::isActive)
        .collect(ArrayList::new, ArrayList::add, ArrayList::addAll);
  }

  /**
   * Updates an existing user.
   * 
   * @param user the user to update
   * @return the updated user
   * @throws IllegalArgumentException if user doesn't exist
   */
  public User update(User user) {
    if (user == null || user.getId() == null) {
      throw new IllegalArgumentException("User and ID cannot be null");
    }

    if (!users.containsKey(user.getId())) {
      throw new IllegalArgumentException("User not found with ID: " + user.getId());
    }

    users.put(user.getId(), user);
    return user;
  }

  /**
   * Deletes a user by ID.
   * 
   * @param id the user ID to delete
   * @return true if user was deleted, false if not found
   */
  public boolean deleteById(Long id) {
    if (id == null) {
      return false;
    }
    return users.remove(id) != null;
  }

  /**
   * Checks if a user exists by username.
   * 
   * @param username the username to check
   * @return true if user exists, false otherwise
   */
  public boolean existsByUsername(String username) {
    if (username == null) {
      return false;
    }

    return users.values().stream()
        .anyMatch(user -> username.equals(user.getUsername()));
  }

  /**
   * Checks if a user exists by email.
   * 
   * @param email the email to check
   * @return true if user exists, false otherwise
   */
  public boolean existsByEmail(String email) {
    if (email == null) {
      return false;
    }

    return users.values().stream()
        .anyMatch(user -> email.equals(user.getEmail()));
  }

  /**
   * Returns the total number of users.
   * 
   * @return total user count
   */
  public long count() {
    return users.size();
  }

  /**
   * Returns the number of active users.
   * 
   * @return active user count
   */
  public long countActive() {
    return users.values().stream()
        .filter(User::isActive)
        .count();
  }

  /**
   * Clears all users from the repository.
   * Useful for testing cleanup.
   */
  public void deleteAll() {
    users.clear();
    idGenerator.set(1);
  }

  /**
   * Simulates a database connection check.
   * Uses the injected DatabaseConfig.
   * 
   * @return true if database connection is valid
   */
  public boolean isConnectionValid() {
    return databaseConfig != null && databaseConfig.isValidConfiguration();
  }

  /**
   * Gets the current database configuration.
   * 
   * @return the database configuration
   */
  public DatabaseConfig getDatabaseConfig() {
    return databaseConfig;
  }
}
