package com.example.observer.eventos;

import com.example.model.transacccion.Transaccion;
import com.example.observer.core.DomainEvent;

public class TransaccionRegistradaEvent extends DomainEvent {
    private final Transaccion transaccion;

    public TransaccionRegistradaEvent(Transaccion transaccion) {
        this.transaccion = transaccion;
    }

    public Transaccion getTransaccion() {
        return transaccion;
    }

    @Override
    public String tipo() {
        return "TRANSACCION_REGISTRADA";
    }
}
