package com.example.model;

import lombok.Data;
import lombok.NoArgsConstructor;
import lombok.AllArgsConstructor;
import lombok.Builder;
import lombok.EqualsAndHashCode;

import java.time.LocalDateTime;

/**
 * User model class representing a user in the system.
 * Uses Lombok annotations for automatic code generation:
 * - @Data: generates getters, setters, equals, hashCode, toString
 * - @Builder: generates builder pattern implementation
 * - @NoArgsConstructor: generates default constructor
 * - @AllArgsConstructor: generates constructor with all fields
 * - @EqualsAndHashCode: customizes equals/hashCode generation
 */
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
@EqualsAndHashCode(exclude = { "createdAt" })
public class User {
  private Long id;
  private String username;
  private String email;

  @Builder.Default
  private LocalDateTime createdAt = LocalDateTime.now();

  @Builder.Default
  private boolean active = true;

  // Constructor for convenience (username, email)
  public User(String username, String email) {
    this.username = username;
    this.email = email;
    this.createdAt = LocalDateTime.now();
    this.active = true;
  }

  // Constructor for convenience (id, username, email)
  public User(Long id, String username, String email) {
    this.id = id;
    this.username = username;
    this.email = email;
    this.createdAt = LocalDateTime.now();
    this.active = true;
  }

  // Business methods
  public void deactivate() {
    this.active = false;
  }

  public void activate() {
    this.active = true;
  }

  public boolean isValidEmail() {
    if (email == null || email.trim().isEmpty()) {
      return false;
    }

    String trimmedEmail = email.trim();

    // Basic email validation: must contain @ and . in proper positions
    int atIndex = trimmedEmail.indexOf('@');
    int lastDotIndex = trimmedEmail.lastIndexOf('.');

    return atIndex > 0 && // @ not at the beginning
        lastDotIndex > atIndex + 1 && // . after @ with at least one character between
        lastDotIndex < trimmedEmail.length() - 1; // . not at the end
  }
}
