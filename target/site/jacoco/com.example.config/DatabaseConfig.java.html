package com.example.config;

import java.util.Properties;

/**
 * DatabaseConfig class implementing the Singleton pattern.
 * This class manages database configuration throughout the application.
 * 
 * Key Singleton characteristics demonstrated:
 * - Private constructor to prevent direct instantiation
 * - Static instance variable
 * - Thread-safe getInstance() method using synchronized keyword
 * - Lazy initialization
 */
public class DatabaseConfig {

  // Static variable to hold the single instance
  private static DatabaseConfig instance;

  // Configuration properties
  private final Properties properties;
  private final String databaseUrl;
  private final String username;
  private final String password;
  private final int maxConnections;

  /**
   * Private constructor to prevent direct instantiation.
   * Initializes default database configuration.
   */
  private DatabaseConfig() {
    this.properties = new Properties();
    this.databaseUrl = "jdbc:h2:mem:testdb";
    this.username = "sa";
    this.password = "";
    this.maxConnections = 10;

    // Load default properties
    loadDefaultProperties();
  }

  /**
   * Thread-safe method to get the singleton instance.
   * Uses synchronized keyword to ensure thread safety.
   * 
   * @return the single instance of DatabaseConfig
   */
  public static synchronized DatabaseConfig getInstance() {
    if (instance == null) {
      instance = new DatabaseConfig();
    }
    return instance;
  }

  /**
   * Alternative thread-safe implementation using double-checked locking.
   * This version is more efficient for high-concurrency scenarios.
   * 
   * @return the single instance of DatabaseConfig
   */
  public static DatabaseConfig getInstanceDoubleChecked() {
    if (instance == null) {
      synchronized (DatabaseConfig.class) {
        if (instance == null) {
          instance = new DatabaseConfig();
        }
      }
    }
    return instance;
  }

  /**
   * Loads default database properties.
   */
  private void loadDefaultProperties() {
    properties.setProperty("database.driver", "org.h2.Driver");
    properties.setProperty("database.url", databaseUrl);
    properties.setProperty("database.username", username);
    properties.setProperty("database.password", password);
    properties.setProperty("database.maxConnections", String.valueOf(maxConnections));
    properties.setProperty("database.autoCommit", "true");
    properties.setProperty("database.timeout", "30");
  }

  // Getter methods
  public String getDatabaseUrl() {
    return databaseUrl;
  }

  public String getUsername() {
    return username;
  }

  public String getPassword() {
    return password;
  }

  public int getMaxConnections() {
    return maxConnections;
  }

  public String getProperty(String key) {
    if (key == null) {
      return null;
    }
    return properties.getProperty(key);
  }

  public Properties getAllProperties() {
    return new Properties(properties);
  }

  /**
   * Updates a configuration property.
   * 
   * @param key   the property key
   * @param value the property value
   */
  public void setProperty(String key, String value) {
    if (key != null) {
      if (value != null) {
        properties.setProperty(key, value);
      } else {
        properties.remove(key);
      }
    }
  }

  /**
   * Checks if the database is configured for production environment.
   * 
   * @return true if configured for production, false otherwise
   */
  public boolean isProductionEnvironment() {
    return !databaseUrl.contains("h2:mem:");
  }

  /**
   * Gets the connection timeout in seconds.
   * 
   * @return connection timeout
   */
  public int getConnectionTimeout() {
    String timeoutStr = properties.getProperty("database.timeout", "30");
    try {
      return Integer.parseInt(timeoutStr);
    } catch (NumberFormatException e) {
      return 30; // default fallback
    }
  }

  /**
   * Validates the current database configuration.
   * 
   * @return true if configuration is valid, false otherwise
   */
  public boolean isValidConfiguration() {
    return databaseUrl != null && !databaseUrl.isEmpty() &&
        username != null &&
        maxConnections > 0;
  }

  /**
   * Reset method for testing purposes only.
   * This method should not be used in production code.
   */
  public static synchronized void resetForTesting() {
    instance = null;
  }

  @Override
  public String toString() {
    return String.format("DatabaseConfig{url='%s', username='%s', maxConnections=%d}",
        databaseUrl, username, maxConnections);
  }
}
