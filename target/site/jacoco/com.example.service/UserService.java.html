package com.example.service;

import com.example.model.User;
import com.example.repository.UserRepository;
import com.example.config.DatabaseConfig;

import java.util.List;
import java.util.Optional;

/**
 * UserService class providing business logic for user operations.
 * This class will be extensively used to demonstrate different testing
 * strategies.
 * 
 * Features demonstrated:
 * - Dependency injection (constructor-based)
 * - Business logic validation
 * - Integration with repository layer
 * - Error handling and validation
 */
public class UserService {

  private final UserRepository userRepository;
  private final DatabaseConfig databaseConfig;

  /**
   * Constructor with dependency injection.
   * 
   * @param userRepository the user repository dependency
   */
  public UserService(UserRepository userRepository) {
    if (userRepository == null) {
      throw new IllegalArgumentException("UserRepository cannot be null");
    }
    this.userRepository = userRepository;
    this.databaseConfig = DatabaseConfig.getInstance();
  }

  /**
   * Creates a new user with business logic validation.
   * 
   * @param username the username
   * @param email    the email address
   * @return the created user
   * @throws IllegalArgumentException if validation fails
   * @throws UserServiceException     if user creation fails
   */
  public User createUser(String username, String email) {
    // Input validation
    validateUsername(username);
    validateEmail(email);

    // Business logic validation
    if (userRepository.existsByUsername(username)) {
      throw new UserServiceException("Username is already taken: " + username);
    }

    if (userRepository.existsByEmail(email)) {
      throw new UserServiceException("Email is already registered: " + email);
    }

    // Create and save user
    User user = new User(username, email);

    try {
      return userRepository.save(user);
    } catch (Exception e) {
      throw new UserServiceException("Failed to create user: " + e.getMessage(), e);
    }
  }

  /**
   * Retrieves a user by ID.
   * 
   * @param id the user ID
   * @return the user if found
   * @throws UserServiceException if user not found
   */
  public User getUserById(Long id) {
    if (id == null) {
      throw new IllegalArgumentException("User ID cannot be null");
    }

    return userRepository.findById(id)
        .orElseThrow(() -> new UserServiceException("User not found with ID: " + id));
  }

  /**
   * Finds a user by username.
   * 
   * @param username the username
   * @return Optional containing the user if found
   */
  public Optional<User> findUserByUsername(String username) {
    if (username == null || username.trim().isEmpty()) {
      return Optional.empty();
    }

    return userRepository.findByUsername(username.trim());
  }

  /**
   * Finds a user by email.
   * 
   * @param email the email
   * @return Optional containing the user if found
   */
  public Optional<User> findUserByEmail(String email) {
    if (email == null || email.trim().isEmpty()) {
      return Optional.empty();
    }

    return userRepository.findByEmail(email.trim().toLowerCase());
  }

  /**
   * Updates user information.
   * 
   * @param userId      the user ID
   * @param newUsername the new username (optional)
   * @param newEmail    the new email (optional)
   * @return the updated user
   * @throws UserServiceException if update fails
   */
  public User updateUser(Long userId, String newUsername, String newEmail) {
    User existingUser = getUserById(userId);

    // Update username if provided
    if (newUsername != null && !newUsername.trim().isEmpty()) {
      validateUsername(newUsername);

      // Check if new username is already taken by another user
      Optional<User> userWithSameUsername = userRepository.findByUsername(newUsername);
      if (userWithSameUsername.isPresent() && !userWithSameUsername.get().getId().equals(userId)) {
        throw new UserServiceException("Username is already taken: " + newUsername);
      }

      existingUser.setUsername(newUsername.trim());
    }

    // Update email if provided
    if (newEmail != null && !newEmail.trim().isEmpty()) {
      validateEmail(newEmail);

      // Check if new email is already taken by another user
      Optional<User> userWithSameEmail = userRepository.findByEmail(newEmail);
      if (userWithSameEmail.isPresent() && !userWithSameEmail.get().getId().equals(userId)) {
        throw new UserServiceException("Email is already registered: " + newEmail);
      }

      existingUser.setEmail(newEmail.trim().toLowerCase());
    }

    try {
      return userRepository.update(existingUser);
    } catch (Exception e) {
      throw new UserServiceException("Failed to update user: " + e.getMessage(), e);
    }
  }

  /**
   * Deactivates a user account.
   * 
   * @param userId the user ID
   * @return the deactivated user
   */
  public User deactivateUser(Long userId) {
    User user = getUserById(userId);
    user.deactivate();

    try {
      return userRepository.update(user);
    } catch (Exception e) {
      throw new UserServiceException("Failed to deactivate user: " + e.getMessage(), e);
    }
  }

  /**
   * Activates a user account.
   * 
   * @param userId the user ID
   * @return the activated user
   */
  public User activateUser(Long userId) {
    User user = getUserById(userId);
    user.activate();

    try {
      return userRepository.update(user);
    } catch (Exception e) {
      throw new UserServiceException("Failed to activate user: " + e.getMessage(), e);
    }
  }

  /**
   * Deletes a user permanently.
   * 
   * @param userId the user ID
   * @throws UserServiceException if deletion fails
   */
  public void deleteUser(Long userId) {
    // Verify user exists
    getUserById(userId);

    try {
      boolean deleted = userRepository.deleteById(userId);
      if (!deleted) {
        throw new UserServiceException("Failed to delete user with ID: " + userId);
      }
    } catch (Exception e) {
      throw new UserServiceException("Failed to delete user: " + e.getMessage(), e);
    }
  }

  /**
   * Retrieves all users.
   * 
   * @return list of all users
   */
  public List<User> getAllUsers() {
    try {
      return userRepository.findAll();
    } catch (Exception e) {
      throw new UserServiceException("Failed to retrieve users: " + e.getMessage(), e);
    }
  }

  /**
   * Retrieves all active users.
   * 
   * @return list of active users
   */
  public List<User> getAllActiveUsers() {
    try {
      return userRepository.findAllActive();
    } catch (Exception e) {
      throw new UserServiceException("Failed to retrieve active users: " + e.getMessage(), e);
    }
  }

  /**
   * Gets the total number of users.
   * 
   * @return total user count
   */
  public long getTotalUserCount() {
    try {
      return userRepository.count();
    } catch (Exception e) {
      throw new UserServiceException("Failed to count users: " + e.getMessage(), e);
    }
  }

  /**
   * Gets the number of active users.
   * 
   * @return active user count
   */
  public long getActiveUserCount() {
    try {
      return userRepository.countActive();
    } catch (Exception e) {
      throw new UserServiceException("Failed to count active users: " + e.getMessage(), e);
    }
  }

  /**
   * Checks if the service is operational.
   * 
   * @return true if service is operational
   */
  public boolean isServiceOperational() {
    try {
      return userRepository.isConnectionValid() &&
          databaseConfig.isValidConfiguration();
    } catch (Exception e) {
      return false;
    }
  }

  // Private validation methods

  private void validateUsername(String username) {
    if (username == null || username.trim().isEmpty()) {
      throw new IllegalArgumentException("Username cannot be null or empty");
    }

    String trimmedUsername = username.trim();
    if (trimmedUsername.length() < 3) {
      throw new IllegalArgumentException("Username must be at least 3 characters long");
    }

    if (trimmedUsername.length() > 50) {
      throw new IllegalArgumentException("Username cannot be longer than 50 characters");
    }

    if (!trimmedUsername.matches("^[a-zA-Z0-9_-]+$")) {
      throw new IllegalArgumentException("Username can only contain letters, numbers, underscores, and hyphens");
    }
  }

  private void validateEmail(String email) {
    if (email == null || email.trim().isEmpty()) {
      throw new IllegalArgumentException("Email cannot be null or empty");
    }

    String trimmedEmail = email.trim();
    if (!trimmedEmail.contains("@") || !trimmedEmail.contains(".")) {
      throw new IllegalArgumentException("Invalid email format");
    }

    if (trimmedEmail.length() > 255) {
      throw new IllegalArgumentException("Email cannot be longer than 255 characters");
    }

    // Basic email validation regex
    String emailRegex = "^[A-Za-z0-9+_.-]+@[A-Za-z0-9.-]+\\.[A-Za-z]{2,}$";
    if (!trimmedEmail.matches(emailRegex)) {
      throw new IllegalArgumentException("Invalid email format");
    }
  }

  /**
   * Custom exception for UserService operations.
   */
  public static class UserServiceException extends RuntimeException {
    public UserServiceException(String message) {
      super(message);
    }

    public UserServiceException(String message, Throwable cause) {
      super(message, cause);
    }
  }
}
